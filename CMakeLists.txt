cmake_minimum_required(VERSION 3.20)
project(bjc85_modern VERSION 0.1.0 LANGUAGES C)
if(NOT APPLE)
  message(FATAL_ERROR "This build targets native macOS ARM64.")
endif()
set(CMAKE_OSX_ARCHITECTURES arm64 CACHE STRING "Native target" FORCE)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_path(LIBUSB_INCLUDE_DIR libusb.h PATHS /opt/homebrew/opt/libusb/include/libusb-1.0 REQUIRED)
find_library(LIBUSB_LIBRARY usb-1.0 PATHS /opt/homebrew/opt/libusb/lib REQUIRED)
add_library(bjc85_transport src/usb.c src/protocol.c src/transfer.c)
target_include_directories(bjc85_transport PUBLIC src ${LIBUSB_INCLUDE_DIR})
target_link_libraries(bjc85_transport PUBLIC ${LIBUSB_LIBRARY})
target_compile_options(bjc85_transport PRIVATE -Wall -Wextra -Werror)
add_executable(bjc85-usb src/usb_cli.c)
target_link_libraries(bjc85-usb PRIVATE bjc85_transport)
target_compile_options(bjc85-usb PRIVATE -Wall -Wextra -Werror)
enable_testing()
add_executable(protocol-tests tests/protocol_test.c src/protocol.c)
target_include_directories(protocol-tests PRIVATE src)
add_test(NAME protocol COMMAND protocol-tests)
add_executable(transfer-tests tests/transfer_test.c src/transfer.c)
target_include_directories(transfer-tests PRIVATE src ${LIBUSB_INCLUDE_DIR})
add_test(NAME transfer COMMAND transfer-tests)
add_library(is12_protocol src/is12_protocol.c)
target_include_directories(is12_protocol PUBLIC src)
target_compile_options(is12_protocol PRIVATE -Wall -Wextra -Werror)
add_executable(is12-protocol-tests tests/is12_protocol_test.c)
target_link_libraries(is12-protocol-tests PRIVATE is12_protocol)
add_test(NAME is12-protocol COMMAND is12-protocol-tests)
add_executable(is12-stream-tests tests/is12_stream_test.c)
target_link_libraries(is12-stream-tests PRIVATE is12_protocol)
target_compile_options(is12-stream-tests PRIVATE -Wall -Wextra -Werror)
add_test(NAME is12-stream COMMAND is12-stream-tests
  "${CMAKE_SOURCE_DIR}/tests/fixtures/is12-first-status.usb"
  "${CMAKE_SOURCE_DIR}/tests/fixtures/is12-status-after-warmup.usb"
  "${CMAKE_SOURCE_DIR}/tests/fixtures/is12-extended-status.usb")
add_library(is12_image src/is12_image.c)
target_link_libraries(is12_image PUBLIC is12_protocol)
target_compile_options(is12_image PRIVATE -Wall -Wextra -Werror)
add_executable(is12-image-tests tests/is12_image_test.c)
target_link_libraries(is12-image-tests PRIVATE is12_image)
target_compile_options(is12-image-tests PRIVATE -Wall -Wextra -Werror)
add_test(NAME is12-image COMMAND is12-image-tests)
add_library(is12_calibration src/is12_calibration.c)
target_include_directories(is12_calibration PUBLIC src)
target_link_libraries(is12_calibration PUBLIC z)
target_compile_options(is12_calibration PRIVATE -Wall -Wextra -Werror)
add_executable(is12-calibration-tests tests/is12_calibration_test.c)
target_link_libraries(is12-calibration-tests PRIVATE is12_calibration)
target_compile_options(is12-calibration-tests PRIVATE -Wall -Wextra -Werror)
add_test(NAME is12-calibration COMMAND is12-calibration-tests
  "${CMAKE_SOURCE_DIR}/tests/fixtures/is12-white-paper-measurement.bin"
  "${CMAKE_SOURCE_DIR}/tests/fixtures/is12-correction-90dpi-temp55.bin"
  "${CMAKE_SOURCE_DIR}/tests/fixtures/is12-correction-180dpi-temp56.bin")
add_library(is12_preview src/is12_preview.c)
target_link_libraries(is12_preview PUBLIC is12_image)
target_compile_options(is12_preview PRIVATE -Wall -Wextra -Werror)
add_executable(is12-preview-replay tests/preview_replay.c)
target_link_libraries(is12-preview-replay PRIVATE is12_preview)
target_compile_options(is12-preview-replay PRIVATE -Wall -Wextra -Werror)
add_executable(bjc85-is12 src/is12_cli.c src/is12_scan.c src/is12_png.c)
target_link_libraries(bjc85-is12 PRIVATE is12_preview is12_calibration bjc85_transport
  "-framework CoreFoundation" "-framework CoreGraphics" "-framework ImageIO")
target_compile_options(bjc85-is12 PRIVATE -Wall -Wextra -Werror)
set(GUTENPRINT_ROOT "${CMAKE_SOURCE_DIR}/.local" CACHE PATH "Locally built Gutenprint")
if(EXISTS "${GUTENPRINT_ROOT}/lib/libgutenprint.a")
  add_executable(bjc85-render src/render.c)
  target_include_directories(bjc85-render PRIVATE "${GUTENPRINT_ROOT}/include")
  target_link_libraries(bjc85-render PRIVATE "${GUTENPRINT_ROOT}/lib/libgutenprint.a"
    "-framework CoreGraphics" "-framework CoreFoundation" "-framework CoreText" "-framework ImageIO")
  target_compile_options(bjc85-render PRIVATE -Wall -Wextra -Werror)
  add_test(NAME gutenprint-bjc85 COMMAND bjc85-render capabilities)
endif()
if(EXISTS "${GUTENPRINT_ROOT}/lib/libgutenprint.a" AND EXISTS "${GUTENPRINT_ROOT}/lib/libpappl.a")
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(OPENSSL REQUIRED IMPORTED_TARGET openssl)
  add_executable(bjc85-ipp src/ipp.c)
  target_include_directories(bjc85-ipp PRIVATE "${GUTENPRINT_ROOT}/include")
  target_link_libraries(bjc85-ipp PRIVATE bjc85_transport "${GUTENPRINT_ROOT}/lib/libgutenprint.a"
    "${GUTENPRINT_ROOT}/lib/libpappl.a" PkgConfig::OPENSSL cups z
    "-framework CoreFoundation" "-framework SystemConfiguration" "-framework AppKit")
  target_compile_options(bjc85-ipp PRIVATE -Wall -Wextra -Werror)
endif()
